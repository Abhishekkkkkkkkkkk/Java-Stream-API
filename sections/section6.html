Collectors Class — Deep Dive

The most powerful class for stream reduction — grouping, partitioning, joining, and more

The java.util.stream.Collectors utility class contains factory methods returning Collector implementations. A Collector encapsulates how to accumulate stream elements into a result container via four components: Supplier, Accumulator, Combiner, and Finisher.

groupingBy() — The Most Powerful Collector

Java — groupingBy examples 
record Employee(String name, String dept, double salary, int age) {}

List<Employee> employees = List.of(
    new Employee("Alice","IT",     90000,30),
    new Employee("Bob",  "Finance",75000,35),
    new Employee("Carol","IT",     85000,28),
    new Employee("Dave", "Finance",95000,40),
    new Employee("Eve",  "HR",     60000,25)
);

// 1. Simple grouping → Map<K, List<T>>
Map<String,List<Employee>> byDept =
    employees.stream().collect(Collectors.groupingBy(Employee::dept));
// {IT=[Alice,Carol], Finance=[Bob,Dave], HR=[Eve]}

// 2. Grouping + count per group (downstream collector)
Map<String,Long> countByDept = employees.stream().collect(
    Collectors.groupingBy(Employee::dept, Collectors.counting()));
// {IT=2, Finance=2, HR=1}

// 3. Grouping + average salary
Map<String,Double> avgSalByDept = employees.stream().collect(
    Collectors.groupingBy(Employee::dept,
        Collectors.averagingDouble(Employee::salary)));
// {IT=87500.0, Finance=85000.0, HR=60000.0}

// 4. Grouping + map to names only (mapping downstream)
Map<String,List<String>> namesByDept = employees.stream().collect(
    Collectors.groupingBy(Employee::dept,
        Collectors.mapping(Employee::name, Collectors.toList())));
// {IT=[Alice,Carol], Finance=[Bob,Dave], HR=[Eve]}

// 5. NESTED grouping (multi-level)
Map<String,Map<String,List<Employee>>> nested = employees.stream().collect(
    Collectors.groupingBy(Employee::dept,
        Collectors.groupingBy(e -> e.age()>30 ? "Senior" : "Junior")));
// {IT={Junior=[Alice,Carol]}, Finance={Senior=[Bob,Dave]}, ...}

// 6. Into a sorted Map (TreeMap)
TreeMap<String,Long> sorted = employees.stream().collect(
    Collectors.groupingBy(Employee::dept, TreeMap::new, Collectors.counting()));

// 7. Top earner per department
Map<String,Optional<Employee>> topEarner = employees.stream().collect(
    Collectors.groupingBy(Employee::dept,
        Collectors.maxBy(Comparator.comparingDouble(Employee::salary))));

partitioningBy()

Java 
// Splits into exactly two groups: true and false
Map<Boolean,List<Employee>> partitioned = employees.stream().collect(
    Collectors.partitioningBy(e -> e.salary() > 80000));
// {true=[Alice,Carol,Dave], false=[Bob,Eve]}

// With downstream collector
Map<Boolean,Long> counted = employees.stream().collect(
    Collectors.partitioningBy(e -> e.salary() > 80000, Collectors.counting()));
// {true=3, false=2}

joining()

Java 
List<String> words = List.of("Java","Stream","API");

words.stream().collect(Collectors.joining());           // "JavaStreamAPI"
words.stream().collect(Collectors.joining(", "));        // "Java, Stream, API"
words.stream().collect(Collectors.joining(", ","[","]"));  // "[Java, Stream, API]"

// Build SQL IN clause
String inClause = List.of("'Alice'","'Bob'").stream()
    .collect(Collectors.joining(",","(",")")); // "('Alice','Bob')"

Statistics Collectors

Java 
List<Integer> scores = List.of(70,85,90,55,78);

long   count = scores.stream().collect(Collectors.counting());           // 5
int    sum   = scores.stream().collect(Collectors.summingInt(Integer::intValue)); // 378
double avg   = scores.stream().collect(Collectors.averagingInt(Integer::intValue)); // 75.6

// All stats at once!
IntSummaryStatistics stats = scores.stream()
    .collect(Collectors.summarizingInt(Integer::intValue));
stats.getCount();   // 5
stats.getSum();     // 378
stats.getMin();     // 55
stats.getMax();     // 90
stats.getAverage(); // 75.6

collectingAndThen() — Post-Process After Collecting

Java 
// Collect to list, then make unmodifiable
List<String> immutable = Stream.of("a","b","c").collect(
    Collectors.collectingAndThen(Collectors.toList(), Collections::unmodifiableList));

// Count then convert to descriptive string
String msg = Stream.of(1,2,3).collect(
    Collectors.collectingAndThen(Collectors.counting(), n -> "Total: "+n));
// "Total: 3"

toMap() with Merge Function

Java 
// ❌ Throws IllegalStateException on duplicate keys
List.of("a","a","b").stream()
    .collect(Collectors.toMap(Function.identity(), String::length));

// ✅ Add merge function: keep first value on collision
List.of("a","a","b").stream()
    .collect(Collectors.toMap(
        Function.identity(),
        String::length,
        (v1, v2) -> v1)); // keep first

// ✅ Sum lengths on duplicate key
List.of("a","aa","b").stream()
    .collect(Collectors.toMap(s -> s.charAt(0), String::length, Integer::sum));
