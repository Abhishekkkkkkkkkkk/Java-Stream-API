Common Stream Patterns

Real-world recipes you'll use every day in backend development

Java — Essential Patterns 
// ─── PATTERN 1: Frequency Count ───────────────────────────────
Map<String,Long> freq = words.stream()
    .collect(Collectors.groupingBy(Function.identity(), Collectors.counting()));

// ─── PATTERN 2: Top N by Field ────────────────────────────────
employees.stream()
    .sorted(Comparator.comparingDouble(Employee::salary).reversed())
    .limit(3)
    .collect(Collectors.toList());

// ─── PATTERN 3: Second Highest Value ─────────────────────────
employees.stream()
    .map(Employee::salary)
    .distinct()
    .sorted(Comparator.reverseOrder())
    .skip(1)
    .findFirst()
    .orElseThrow();

// ─── PATTERN 4: Group + Join Names ───────────────────────────
Map<String,String> deptNames = employees.stream()
    .collect(Collectors.groupingBy(Employee::dept,
        Collectors.mapping(Employee::name, Collectors.joining(", "))));
// {IT="Alice, Carol", Finance="Bob, Dave", HR="Eve"}

// ─── PATTERN 5: List → Map (id → object) ─────────────────────
Map<Integer,Employee> empById = employees.stream()
    .collect(Collectors.toMap(Employee::id, Function.identity()));

// ─── PATTERN 6: Flatten Nested List ──────────────────────────
List<Integer> flat = nested.stream()
    .flatMap(Collection::stream)
    .distinct().sorted().collect(Collectors.toList());

// ─── PATTERN 7: Dedup by Custom Key ──────────────────────────
employees.stream()
    .collect(Collectors.toMap(Employee::dept, e -> e, (a,b) -> a))
    .values().stream();

// ─── PATTERN 8: Pagination ────────────────────────────────────
int page=2, pageSize=10;
list.stream().skip((long)(page-1)*pageSize).limit(pageSize).collect(Collectors.toList());

// ─── PATTERN 9: Merge Two Streams ────────────────────────────
Stream.concat(stream1, stream2).distinct().sorted().collect(Collectors.toList());

// ─── PATTERN 10: Partition then Aggregate ────────────────────
Map<Boolean,DoubleSummaryStatistics> salStats = employees.stream()
    .collect(Collectors.partitioningBy(e -> e.salary()>80000,
        Collectors.summarizingDouble(Employee::salary)));

// ─── PATTERN 11: Running Total ────────────────────────────────
AtomicInteger running = new AtomicInteger();
List<Integer> cumSum = List.of(1,2,3,4,5).stream()
    .map(running::addAndGet).collect(Collectors.toList());
// [1,3,6,10,15]

// ─── PATTERN 12: Log Analysis from File ───────────────────────
try (Stream<String> lines = Files.lines(Path.of("app.log"))) {
    Map<String,Long> errorCounts = lines
        .filter(l -> l.contains("ERROR"))
        .map(l -> l.split(" ")[2])
        .collect(Collectors.groupingBy(Function.identity(), Collectors.counting()));
}
